pragma solidity ^0.8.0;

contract StudentData {

    struct Student {
        uint rollNo;
        string name;
    }

    Student[] public students;

    function addStudent(uint r, string memory n) public {
        students.push(Student(r, n));
    }

    function getStudent(uint i) public view returns(uint, string memory) {
        Student memory s = students[i];
        return (s.rollNo, s.name);
    }
    receive() external payable{}
    fallback() external payable {}
}
